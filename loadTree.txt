    function handleUploadSelect(files) {
      const allowed = ['.pdf','.doc','.docx','.xls','.xlsx','.odt','.ods','.rtf','.dwg','.dxf','.jpg','.jpeg','.png','.tiff','.tif'];

      for (const f of files) {

        const ext = name.includes('.') ? '.' + name.split('.').pop() : '';
        if (allowed.includes(ext)) {

        } else {

    }

    async function getFilesFromDataTransfer(dataTransfer) {
      if (!dataTransfer.items) {
        return Array.from(dataTransfer.files || []);
      }
      const files = [];
      const promises = [];
      for (let i = 0; i < dataTransfer.items.length; i++) {
        const item = dataTransfer.items[i];
        if (item.kind === 'file') {
          const entry = item.webkitGetAsEntry();
          if (entry) {
            promises.push(traverseFileTree(entry, '', files));
          }
        }
      }
      await Promise.all(promises);
      return files;
    }

    function traverseFileTree(item, path, filesList) {
      path = path || "";
      if (item.isFile) {
        return new Promise((resolve) => {
          item.file((file) => {
            file._relativePath = path + file.name;
            filesList.push(file);
            resolve();
          });
        });
      } else if (item.isDirectory) {
        return new Promise((resolve) => {
          const dirReader = item.createReader();
          const entries = [];
          const readEntries = () => {
            dirReader.readEntries(async (results) => {
              if (!results.length) {
                const promises = [];
                for (const entry of entries) {
                  promises.push(traverseFileTree(entry, path + item.name + "/", filesList));
                }
                await Promise.all(promises);
                resolve();
              } else {
                entries.push(...results);
                readEntries();
              }
            });
          };
          readEntries();
        });
      }
    }

    function formatJobTime(ts) {
      if (!ts) return '';
      try {
        return new Date(ts * 1000).toLocaleString('ru-RU', { day: '2-digit', month: '2-digit', hour: '2-digit', minute: '2-digit' });

    function formatDuration(started, finished) {
      if (!started || !finished) return "";
      const secs = Math.round(finished - started);
      if (secs <= 0) return "";
      const m = Math.floor(secs / 60), s = secs % 60;
      return m > 0 ? (m + " РјРёРЅ " + s + " СЃРµРє") : (s + " СЃРµРє");
    }

      if (st === 'cancelled') return 'вЉ РѕС‚РјРµРЅРµРЅРѕ';

    function jobFilesInfo(job) {
      const meta = job.meta || {};
      const summary = job.result_summary || {};
      const doneFiles = Array.isArray(summary.files) ? summary.files : null;
      if (doneFiles && doneFiles.length) {
        return {
          files: doneFiles,
          kind: 'result',
          total: doneFiles.length,
          truncated: !!summary.files_truncated,
        };
      }
      const previewFiles = Array.isArray(meta.files) ? meta.files : null;
      if (previewFiles && previewFiles.length) {
        return {
          files: previewFiles,
          kind: 'preview',
          total: meta.files_count || previewFiles.length,
          truncated: !!meta.truncated,
        };
      }
      return null;
    }

    function escapeHtml(s) {
      return String(s || '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function renderJobFiles(job) {
      const info = jobFilesInfo(job);
      if (!info) return '';
      const head = info.kind === 'result'
        ? ('РћР±СЂР°Р±РѕС‚Р°РЅРѕ С„Р°Р№Р»РѕРІ: ' + info.total + (info.truncated ? ' (РїРѕРєР°Р·Р°РЅС‹ РїРµСЂРІС‹Рµ 500)' : ''))
        : ('Рљ РїРµСЂРµРІРѕРґСѓ: ' + info.total + (info.truncated ? ' (РїРѕРєР°Р·Р°РЅС‹ РїРµСЂРІС‹Рµ 500)' : ''));
      const rows = info.files.map(f => {
        const name = f.name || (f.path || '').split('/').pop() || f.path || 'вЂ”';
        let meta = f.format || '';
        let cls = '';
        if (info.kind === 'result') {
          const st = (f.status || '').toLowerCase();
          meta = f.status || '';
          if (f.message) meta += (meta ? ': ' : '') + f.message;
          if (st === 'ok' || st === 'done') cls = ' ok';
          else if (st === 'error' || st === 'fail') cls = ' error';
          else if (st === 'skip' || st === 'skipped') cls = ' skip';
        }
        return '<div class="job-file-row' + cls + '">' +
          '<span class="job-file-name" title="' + escapeHtml(f.path || name) + '">' + escapeHtml(name) + '</span>' +
          '<span class="job-file-meta" title="' + escapeHtml(meta) + '">' + escapeHtml(meta) + '</span>' +
          '</div>';
      }).join('');
      return '<div class="job-files-head">' + escapeHtml(head) + '</div>' + rows;
    }

    function formatSmbConnected(mount) {
      if (!mount) return '';
      const host = mount.server || (mount.unc || '').replace(/^\/\//, '').split('/')[0] || '';
      const share = mount.share_label || mount.share || (mount.unc || '').split('/').filter(Boolean)[1] || '';
      if (host && share) return host + ' в†’ ' + share;
      return host || share || '';
    }

    function portalApiBases() {
      const p = window.location.port;
      if (!p || p === '80' || p === '443' || p === '8080') return [''];
      const { protocol, hostname } = window.location;
      if (p === '8084') return [`${protocol}//${hostname}:8080`, `${protocol}//${hostname}`];
      return [`${protocol}//${hostname}`];
    }
    async function portalFetch(path, init) {
      let lastErr = null;
      for (const base of portalApiBases()) {
        try {
          const r = await fetch(base + path, init);
          if (r.status === 404 && portalApiBases().length > 1 && base === portalApiBases()[0]) continue;
          return r;
        } catch (e) {
          lastErr = e;
        }
      }
      throw lastErr || new Error('РџРѕСЂС‚Р°Р» РЅРµРґРѕСЃС‚СѓРїРµРЅ. РћС‚РєСЂРѕР№С‚Рµ СЃС‚СЂР°РЅРёС†Сѓ С‡РµСЂРµР· /convert РЅР° РїРѕСЂС‚Сѓ 80 РёР»Рё 8080.');
